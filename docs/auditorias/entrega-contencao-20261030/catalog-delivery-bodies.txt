  v_motivo_nb text := nullif(btrim(coalesce(p_opcoes->>'motivo_nao_faturavel','')), '');
BEGIN

  IF p_entrega_id IS NULL THEN RAISE EXCEPTION 'COCKPIT_ENTREGA_PARAMS_INVALIDOS'; END IF;

  PERFORM public.fn_estoque_ctx_open(jsonb_build_object(
    'acao','entrega','origem','fn_entrega_confirmar_engine_v1','usuario_id', v_uid::text));

  PERFORM pg_advisory_xact_lock(hashtext('cockpit_entrega:' || p_entrega_id::text));

  SELECT e.id, e.status::text AS status, e.estoque_produto_id,
         coalesce(e.quantidade_entregue, 0) AS qtd, e.partner_order_id,
         e.ordem_servico_id, e.ordem_servico_onfarm_id
    INTO v_ent
    FROM public.entregas e
   WHERE e.id = p_entrega_id
   FOR UPDATE;

  IF v_ent.id IS NULL THEN RAISE EXCEPTION 'COCKPIT_ENTREGA_NAO_ENCONTRADA'; END IF;

  -- Origem: explícita quando informada; derivada do vínculo caso contrário.

    v_origem := public.fn_cockpit_norm_origem(p_origem);
  ELSIF v_ent.ordem_servico_onfarm_id IS NOT NULL THEN
    v_origem := 'onfarm';
  ELSIF v_ent.partner_order_id IS NOT NULL THEN
    v_origem := 'b2b';
  ELSE
    v_origem := 'manual';
  END IF;

  v_origem := public.fn_entrega_assert_confirmador(v_origem);

  v_os_id := CASE WHEN v_origem = 'onfarm' THEN v_ent.ordem_servico_onfarm_id
                  ELSE v_ent.ordem_servico_id END;

  IF p_os_id IS NOT NULL AND v_os_id IS DISTINCT FROM p_os_id THEN
    RAISE EXCEPTION 'COCKPIT_RELATION_MISMATCH_ENTREGA_OS';
  END IF;

  IF v_ent.status = 'cancelada' THEN RAISE EXCEPTION 'COCKPIT_ENTREGA_CANCELADA'; END IF;
  IF v_ent.status = 'entregue' THEN
    RETURN jsonb_build_object('success', true, 'replay', true, 'finalizada', false,
      'entrega_id', p_entrega_id,
      'message', 'Entrega já confirmada anteriormente.');
  END IF;

  IF v_origem = 'b2b' AND v_ent.partner_order_id IS NOT NULL THEN
    PERFORM public.fn_cockpit_assert_b2b_liberado(v_ent.partner_order_id);
  END IF;

  IF v_ent.estoque_produto_id IS NOT NULL THEN
    SELECT * INTO v_est FROM public.estoque_produto WHERE id = v_ent.estoque_produto_id FOR UPDATE;
    IF v_est.id IS NULL THEN RAISE EXCEPTION 'COCKPIT_ENTREGA_ESTOQUE_NAO_ENCONTRADO'; END IF;
    IF v_est.status::text <> 'expedido' THEN
      IF v_os_id IS NOT NULL THEN
        IF v_est.reservado_para_os_id IS DISTINCT FROM v_os_id THEN
          RAISE EXCEPTION 'COCKPIT_RELATION_MISMATCH_ENTREGA_ESTOQUE';
        END IF;
        IF coalesce(v_est.reservado_para_os_tipo, v_origem) IS DISTINCT FROM v_origem THEN
          RAISE EXCEPTION 'COCKPIT_RELATION_MISMATCH_ENTREGA_ORIGEM';
        END IF;
      END IF;
      IF v_ent.qtd > coalesce(v_est.quantidade, 0) THEN
        RAISE EXCEPTION 'COCKPIT_ENTREGA_QUANTIDADE_EXCEDE_RESERVA';
      END IF;
    END IF;
  END IF;

  -- Recebedor físico é OBRIGATÓRIO e distinto de confirmado_por.
  v_recebedor_nome := nullif(btrim(coalesce(p_recebedor->>'nome','')), '');
  IF v_recebedor_nome IS NULL THEN
    RAISE EXCEPTION 'COCKPIT_ENTREGA_RECEBEDOR_OBRIGATORIO';
  END IF;

  v_entregue_em := coalesce(nullif(btrim(coalesce(p_opcoes->>'entregue_em','')), '')::timestamptz, now());

     SET status = 'entregue'::public.status_entrega,
         entregue_em = v_entregue_em,
         recebido_por_nome = v_recebedor_nome,
         recebido_por_cargo = coalesce(nullif(btrim(coalesce(p_recebedor->>'cargo','')), ''), recebido_por_cargo),
         recebido_por_documento = coalesce(nullif(btrim(coalesce(p_recebedor->>'documento','')), ''), recebido_por_documento),
         observacoes_confirmacao = coalesce(nullif(btrim(coalesce(p_recebedor->>'observacoes','')), ''), observacoes_confirmacao),
         confirmado_por = v_uid,
         financial_status = CASE WHEN v_faturamento = 'bill_later' THEN 'PENDING_BILLING'
                                 ELSE financial_status END
   WHERE id = p_entrega_id;

  IF v_est.id IS NOT NULL THEN
    UPDATE public.estoque_produto
       SET status = 'expedido'::public.status_estoque, expedido_por = v_uid,
           expedido_em = now(), updated_at = now()
     WHERE id = v_est.id AND status <> 'expedido'::public.status_estoque;

    IF FOUND THEN
      INSERT INTO public.estoque_movimentacoes
        (estoque_produto_id, acao, quantidade, status_anterior, status_novo, os_id, os_tipo, usuario_id, lote_id, observacao)
      VALUES (v_est.id, 'expedicao', v_ent.qtd, 'reservado', 'expedido',
              v_os_id, v_origem, v_uid, v_est.lote_id, 'Baixa por confirmação de entrega');
    END IF;
  END IF;

  IF v_os_id IS NOT NULL THEN
    v_avanco := public._rpc_avancar_esteira_laboratorio_internal(v_os_id, v_origem, 'enviar_entrega', NULL);

    INSERT INTO public.laboratorio_esteira_eventos
      (os_id, origem_os, etapa_anterior, etapa_nova, acao, usuario_id, observacao)
    VALUES (v_os_id, v_origem, 'em_entrega', 'entregue', 'confirmar_entrega_v1', v_uid,
            format('Confirmação transacional da entrega %s (recebedor: %s)', p_entrega_id, v_recebedor_nome));

    IF v_origem = 'onfarm' THEN
      SELECT count(*) FILTER (WHERE status::text NOT IN ('entregue','cancelada')),
             coalesce(sum(quantidade_entregue) FILTER (WHERE status::text = 'entregue'), 0)
        INTO v_pendentes, v_total_entregue
        FROM public.entregas WHERE ordem_servico_onfarm_id = v_os_id;
    ELSE
      SELECT count(*) FILTER (WHERE status::text NOT IN ('entregue','cancelada')),
             coalesce(sum(quantidade_entregue) FILTER (WHERE status::text = 'entregue'), 0)
        INTO v_pendentes, v_total_entregue
        FROM public.entregas WHERE ordem_servico_id = v_os_id;
    END IF;

    SELECT coalesce(sum(quantidade_reservada), 0) INTO v_total_reservado
      FROM public.estoque_produto
     WHERE reservado_para_os_id = v_os_id
       AND coalesce(reservado_para_os_tipo, v_origem) = v_origem
       AND status <> 'estornado'::public.status_estoque;

    IF v_pendentes = 0 AND v_total_reservado > 0 AND v_total_entregue >= v_total_reservado THEN
      v_avanco := public._rpc_avancar_esteira_laboratorio_internal(v_os_id, v_origem, 'finalizar_os', NULL);
      IF coalesce((v_avanco->>'success')::boolean, false) THEN
        v_finalizada := true;
        INSERT INTO public.laboratorio_esteira_eventos
          (os_id, origem_os, etapa_anterior, etapa_nova, acao, usuario_id, observacao)
        VALUES (v_os_id, v_origem, 'entregue', 'finalizada', 'finalizacao_automatica_por_entrega', v_uid,
                'Cobertura completa: OS finalizada na mesma transação da confirmação de entrega.');
      END IF;
    END IF;
  END IF;

  IF v_faturamento = 'not_billable' THEN
    IF v_motivo_nb IS NULL THEN RAISE EXCEPTION 'COCKPIT_ENTREGA_MOTIVO_NAO_FATURAVEL_OBRIGATORIO'; END IF;
    PERFORM public.fn_mark_entrega_not_billable(p_entrega_id, v_motivo_nb);
  END IF;

  RETURN jsonb_build_object('success', true, 'replay', false, 'finalizada', v_finalizada,
    'entrega_id', p_entrega_id, 'os_id', v_os_id, 'origem', v_origem,
    'parcial', (coalesce(v_pendentes, 0) > 0 OR coalesce(v_total_entregue, 0) < coalesce(v_total_reservado, 0)),
    'total_entregue', coalesce(v_total_entregue, 0), 'total_reservado', coalesce(v_total_reservado, 0),
    'message', CASE WHEN v_finalizada THEN 'Entrega confirmada — OS finalizada.'
                    ELSE 'Entrega confirmada.' END);
END;

 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_e public.entregas%ROWTYPE;
  v_po public.partner_orders%ROWTYPE;
  v_t public.partner_order_payment_terms%ROWTYPE;
  v_fin text;
  v_ap text;
  v_total_snapshot numeric;
  v_origem text;
  v_base numeric;
  v_aceito numeric;
  v_valor numeric;
  v_conf int;
  v_itens jsonb;
  v_ident jsonb;
BEGIN
  SELECT * INTO v_e FROM public.entregas WHERE id = p_entrega_id;
  IF NOT FOUND THEN
    RETURN jsonb_build_object('origem','indisponivel','motivo','entrega_nao_encontrada');
  END IF;

  SELECT count(*)::int INTO v_conf
    FROM public.entrega_conferencia_itens WHERE entrega_id = p_entrega_id;

  -- ------------------- caminho B2B (snapshot canônico) -------------------
  IF v_e.partner_order_id IS NOT NULL THEN
    SELECT * INTO v_po FROM public.partner_orders WHERE id = v_e.partner_order_id;
    IF NOT FOUND THEN
      RETURN jsonb_build_object('origem','indisponivel','motivo','pedido_b2b_nao_encontrado');
    END IF;

    v_fin := COALESCE(v_po.financial_approval_status, 'not_required');
    v_ident := jsonb_build_object(
      'partner_order_id', v_po.id,
      'payment_terms_id', v_po.payment_terms_id,
      'financial_approval_status', v_fin,
      'order_status', v_po.status);

    IF v_po.payment_terms_id IS NOT NULL THEN
      -- identidade imutável da condição econômica
      SELECT * INTO v_t FROM public.partner_order_payment_terms
       WHERE id = v_po.payment_terms_id;
      IF NOT FOUND OR v_t.partner_order_id IS DISTINCT FROM v_po.id THEN
        RETURN jsonb_build_object('origem','bloqueado','motivo','termo_canonico_invalido')
               || v_ident;
      END IF;

      v_ap := COALESCE(v_t.approval_status, 'pending');
      v_ident := v_ident || jsonb_build_object('approval_status', v_ap);

      IF v_ap = 'approved' THEN
        NULL; -- liberado
      ELSIF v_ap = 'not_required'
            AND v_fin = 'not_required'
            AND COALESCE(v_po.requires_financial_approval, false) = false THEN
        NULL; -- liberado: nenhuma aprovação exigida em nenhum nível
      ELSE
        -- pending / rejected / adjustment_requested / incoerência => fail-closed.
        -- Proibido fallback silencioso para outro termo ou para o pedido.
        RETURN jsonb_build_object('origem','bloqueado','motivo','condicao_b2b_nao_liberada')
               || v_ident;
      END IF;

      v_total_snapshot := v_t.final_amount;
      v_origem := 'B2B_PAYMENT_TERMS_SNAPSHOT';
    ELSE
      -- legado: sem condição vinculada ao pedido.
      IF EXISTS (SELECT 1 FROM public.partner_order_payment_terms t
                  WHERE t.partner_order_id = v_po.id) THEN
        -- existe termo, mas não é o canônico do pedido => nunca usar "o mais novo"
        RETURN jsonb_build_object('origem','bloqueado','motivo','termo_nao_vinculado_ao_pedido')
               || v_ident;
      END IF;
      IF v_fin IN ('pending','rejected') THEN
        RETURN jsonb_build_object('origem','bloqueado','motivo','aprovacao_financeira_'||v_fin)
               || v_ident;
      END IF;
      IF COALESCE(v_po.status,'') IN ('cancelado','reprovado','reprovado_sem_itens',
                                      'rascunho','pendente','solicitado','em_analise_comercial',
                                      'aguardando_aprovacao_credito','aguardando_ajuste_comercial',
                                      'aguardando_itens','aguardando_cotacao') THEN
        RETURN jsonb_build_object('origem','bloqueado','motivo','pedido_nao_faturavel')
               || v_ident;
      END IF;
      v_total_snapshot := COALESCE(v_po.final_amount_with_interest, v_po.total);
      v_origem := 'B2B_ORDER_SNAPSHOT_LEGACY';
    END IF;

    IF v_total_snapshot IS NULL OR v_total_snapshot <= 0 THEN
      RETURN jsonb_build_object('origem','indisponivel','motivo','snapshot_b2b_ausente')
             || v_ident;
    END IF;

    -- base econômica do pedido = soma dos subtotais snapshot dos itens
    SELECT COALESCE(sum(COALESCE(i.subtotal,
             COALESCE(i.preco_unitario_final_snapshot, i.preco_unitario, 0) * COALESCE(i.quantidade,0))), 0)
      INTO v_base
      FROM public.partner_order_items i WHERE i.order_id = v_po.id;

    IF v_conf = 0 OR v_base <= 0 THEN
      RETURN jsonb_build_object('origem', v_origem, 'valor', round(v_total_snapshot,2),
        'metodo', jsonb_build_object('tipo','integral','total_snapshot', round(v_total_snapshot,2),
                                     'base_itens', round(COALESCE(v_base,0),2)))
             || v_ident;
    END IF;

    -- participação econômica aceita POR PRODUTO (volume conferido aceito / volume conferido)
    WITH conf AS (
      SELECT ee.produto_id AS produto_id, ci.status AS status,
             COALESCE(ee.volume,0) * COALESCE(ci.quantidade,1) AS vol
        FROM public.entrega_conferencia_itens ci
        JOIN public.estoque_embalagens ee ON ee.id = ci.embalagem_id
       WHERE ci.entrega_id = p_entrega_id
    ), ratio AS (
      SELECT produto_id,
             CASE WHEN SUM(vol) > 0
                  THEN LEAST(1, round(COALESCE(SUM(vol) FILTER (WHERE status='aceito'), 0) / SUM(vol), 6))
                  ELSE 0 END AS r
        FROM conf GROUP BY produto_id
    ), itens AS (
      SELECT i.product_id AS product_id,
             COALESCE(i.subtotal,
               COALESCE(i.preco_unitario_final_snapshot, i.preco_unitario, 0) * COALESCE(i.quantidade,0)) AS sub,
             COALESCE(r.r, 0) AS r
        FROM public.partner_order_items i
        LEFT JOIN ratio r ON r.produto_id = i.product_id
       WHERE i.order_id = v_po.id
    )
    SELECT COALESCE(sum(sub * r),0),
           COALESCE(jsonb_agg(jsonb_build_object('produto_id', product_id,
                     'subtotal_snapshot', round(sub,2), 'ratio_aceito', r,
                     'base_aceita_item', round(sub * r, 2))
                    ORDER BY product_id), '[]'::jsonb)
      INTO v_aceito, v_itens
      FROM itens;

    IF abs(v_aceito - v_base) <= 0.01 THEN
      v_valor := round(v_total_snapshot, 2);
    ELSE
      -- alocação do ajuste histórico do pedido (juros/desconto de nível pedido)
      -- proporcional à participação econômica aceita dos itens.
      v_valor := round(v_total_snapshot * (v_aceito / v_base), 2);
    END IF;

    RETURN jsonb_build_object('origem', v_origem, 'valor', v_valor,
      'metodo', jsonb_build_object('tipo','participacao_economica_aceita_x_snapshot',
        'total_snapshot', round(v_total_snapshot,2),
        'base_itens', round(v_base,2), 'base_aceita', round(v_aceito,2),
        'fator', round(v_aceito / NULLIF(v_base,0), 6), 'itens', v_itens))
           || v_ident;
  END IF;

  -- ---------------- caminho OS / base de faturamento histórica -------------
  IF v_e.os_faturamento_base_id IS NOT NULL THEN
    SELECT b.valor_total_faturavel INTO v_total_snapshot
      FROM public.os_faturamento_base b WHERE b.id = v_e.os_faturamento_base_id;
    IF v_total_snapshot IS NOT NULL AND v_total_snapshot > 0 THEN
      RETURN jsonb_build_object('origem','OS_FATURAMENTO_BASE_SNAPSHOT',
        'valor', round(v_total_snapshot,2),
        'metodo', jsonb_build_object('tipo','os_base','os_faturamento_base_id', v_e.os_faturamento_base_id));
    END IF;
  END IF;

  RETURN jsonb_build_object('origem','indisponivel','motivo','sem_fonte_historica_confiavel');
END

  v_canon_valor numeric;
  v_canon_origem text;
  v_manual boolean;
  v_manual_reason text;
  v_count integer;
  v_category uuid; v_cost_center uuid; v_payment_method uuid;
  v_first_due date; v_issue date;
  v_fiscal_required boolean; v_fiscal_status text;
  v_installments jsonb; v_sum numeric(14,2) := 0; v_inst jsonb;
  v_desc text; v_notes text; v_guard jsonb;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sessão inválida.' USING ERRCODE='insufficient_privilege';
  END IF;
  IF NOT public.is_financial_staff(v_uid) THEN
    RAISE EXCEPTION 'Sem permissão para faturar entregas.' USING ERRCODE='insufficient_privilege';
  END IF;

  PERFORM pg_advisory_xact_lock(hashtextextended('entrega_faturar:' || _entrega_id::text, 0));

  SELECT * INTO v_entrega FROM public.entregas WHERE id = _entrega_id FOR UPDATE;
  IF NOT FOUND THEN RAISE EXCEPTION 'Entrega não encontrada.'; END IF;

  -- H6-003.5 replay estavel
  IF v_entrega.receivable_id IS NOT NULL THEN
    RETURN v_entrega.receivable_id;
  END IF;

  IF v_entrega.status = 'cancelada' THEN
    RAISE EXCEPTION 'Entrega cancelada não pode ser faturada.' USING ERRCODE='check_violation';
  END IF;
  IF v_entrega.status <> 'entregue' THEN
    RAISE EXCEPTION 'Entrega precisa estar com status "entregue" para faturar.' USING ERRCODE='check_violation';
  END IF;
  IF v_entrega.cliente_id IS NULL THEN
    RAISE EXCEPTION 'Entrega sem cliente vinculado.' USING ERRCODE='check_violation';
  END IF;
  IF v_entrega.financial_status = 'NOT_BILLABLE' THEN
    RAISE EXCEPTION 'Entrega marcada como não faturável.' USING ERRCODE='check_violation';
  END IF;

  v_total           := COALESCE((_payload->>'total_amount')::numeric, 0);
  v_count           := COALESCE((_payload->>'installment_count')::integer, 1);
  v_category        := NULLIF(_payload->>'category_id','')::uuid;
  v_cost_center     := NULLIF(_payload->>'cost_center_id','')::uuid;
  v_payment_method  := NULLIF(_payload->>'payment_method_id','')::uuid;
  v_first_due       := COALESCE((_payload->>'first_due_date')::date, CURRENT_DATE);
  v_issue           := COALESCE((_payload->>'issue_date')::date, CURRENT_DATE);
  v_fiscal_required := COALESCE((_payload->>'fiscal_invoice_required')::boolean, false);
  v_fiscal_status   := CASE WHEN v_fiscal_required THEN 'PENDING' ELSE 'NOT_REQUIRED' END;
  v_installments    := _payload->'installments';
  v_desc            := COALESCE(NULLIF(_payload->>'description',''), 'Entrega ' || substr(_entrega_id::text,1,8));
  v_notes           := NULLIF(_payload->>'notes','');
  v_manual          := COALESCE((_payload->>'manual_financial_adjustment')::boolean, false);
  v_manual_reason   := NULLIF(trim(_payload->>'manual_adjustment_reason'), '');

  IF v_category IS NULL THEN
    RAISE EXCEPTION 'Categoria financeira é obrigatória.' USING ERRCODE='check_violation';
  END IF;
  IF NOT EXISTS (SELECT 1 FROM public.financial_categories WHERE id = v_category) THEN
    RAISE EXCEPTION 'Categoria financeira não encontrada.' USING ERRCODE='check_violation';
  END IF;
  IF v_count < 1 OR v_count > 36 THEN
    RAISE EXCEPTION 'Número de parcelas deve estar entre 1 e 36.' USING ERRCODE='check_violation';
  END IF;

  -- ------- H6-003: valor canonico server-side é a autoridade do total -------
  v_canon := public.fn_entrega_valor_faturavel_v1(_entrega_id);
  v_canon_origem := v_canon->>'origem';

  -- reconciliacao parcial: usa exatamente o valor_faturavel_max persistido
  SELECT r.valor_faturavel_max INTO v_canon_valor
    FROM public.entrega_reconciliacoes r
   WHERE r.entrega_id = _entrega_id AND r.status = 'reconciliada'
   LIMIT 1;
  IF v_canon_valor IS NULL THEN
    v_canon_valor := (v_canon->>'valor')::numeric;
  ELSE
    v_canon_origem := 'RECONCILIACAO:' || COALESCE(v_canon_origem,'-');
  END IF;

  IF v_canon_valor IS NOT NULL THEN
    IF abs(v_total - v_canon_valor) > 0.01 THEN
      RAISE EXCEPTION 'valor_divergente_do_canonico: informado %, canônico % (origem %).',
        v_total, v_canon_valor, v_canon_origem USING ERRCODE='check_violation';
    END IF;
    v_total := round(v_canon_valor, 2);
  ELSE
    -- sem fonte historica confiavel: exige ajuste financeiro manual justificado
    IF NOT v_manual OR v_manual_reason IS NULL OR length(v_manual_reason) < 10 THEN
      RAISE EXCEPTION 'valor_canonico_indisponivel: informe manual_financial_adjustment com motivo (mín. 10 caracteres).'
        USING ERRCODE='check_violation';
    END IF;
  END IF;

  IF v_total <= 0 THEN
    RAISE EXCEPTION 'Valor total inválido.' USING ERRCODE='check_violation';
  END IF;

  -- GATE: reconciliação obrigatória quando há conferência por item
  v_guard := public.fn_entrega_guard_faturamento_v1(_entrega_id, v_total);

  IF v_installments IS NOT NULL AND jsonb_array_length(v_installments) > 0 THEN
    IF jsonb_array_length(v_installments) <> v_count THEN
      RAISE EXCEPTION 'Quantidade de parcelas (%) diverge de installment_count (%).',
        jsonb_array_length(v_installments), v_count USING ERRCODE='check_violation';
    END IF;
    FOR v_inst IN SELECT * FROM jsonb_array_elements(v_installments) LOOP
      v_sum := v_sum + COALESCE((v_inst->>'amount')::numeric, 0);
    END LOOP;
    IF abs(v_sum - v_total) > 0.01 THEN
      RAISE EXCEPTION 'Soma das parcelas (%) difere do total (%).', v_sum, v_total USING ERRCODE='check_violation';
    END IF;
  END IF;

  BEGIN
    INSERT INTO public.accounts_receivable (
      origin_type, origin_id, customer_id, description, category_id, cost_center_id,
      total_amount, issue_date, first_due_date, installment_count,
      payment_method_id, fiscal_invoice_required, fiscal_invoice_status,
      notes, created_by
    ) VALUES (
      'DELIVERY', _entrega_id, v_entrega.cliente_id, v_desc, v_category, v_cost_center,
      v_total, v_issue, v_first_due, v_count,
      v_payment_method, v_fiscal_required, v_fiscal_status,
      v_notes, v_uid
    ) RETURNING id INTO v_receivable_id;
  EXCEPTION WHEN unique_violation THEN
    SELECT id INTO v_receivable_id FROM public.accounts_receivable
     WHERE origin_type = 'DELIVERY' AND origin_id = _entrega_id AND status <> 'CANCELED';
    IF v_receivable_id IS NOT NULL THEN RETURN v_receivable_id; END IF;
    RAISE EXCEPTION 'Entrega já possui conta a receber ativa.' USING ERRCODE='check_violation';
  END;

  IF v_installments IS NOT NULL AND jsonb_array_length(v_installments) > 0 THEN
    DELETE FROM public.accounts_receivable_installments WHERE receivable_id = v_receivable_id;
    FOR v_inst IN SELECT * FROM jsonb_array_elements(v_installments) LOOP
      INSERT INTO public.accounts_receivable_installments (
        receivable_id, installment_number, due_date, original_amount, net_amount, status
      ) VALUES (
        v_receivable_id,
        (v_inst->>'installment_number')::integer,
        (v_inst->>'due_date')::date,
        (v_inst->>'amount')::numeric,
        (v_inst->>'amount')::numeric,
        'OPEN'
      );
    END LOOP;
  END IF;

  UPDATE public.entregas
     SET receivable_id = v_receivable_id,
         financial_status = 'BILLED',
         billing_generated_at = now(),
         billing_generated_by = v_uid
   WHERE id = _entrega_id;

  INSERT INTO public.financial_audit_log (
    entity_type, entity_id, action, old_data, new_data, performed_by
  ) VALUES (
    'entregas', _entrega_id, 'DELIVERY_BILLED',
    jsonb_build_object('receivable_id', v_entrega.receivable_id,
                       'financial_status', v_entrega.financial_status),
    jsonb_build_object('receivable_id', v_receivable_id, 'total_amount', v_total,
      'installment_count', v_count, 'fiscal_invoice_required', v_fiscal_required,
      'payment_method_id', v_payment_method, 'origem', 'entrega',
      'valor_canonico', v_canon_valor, 'valor_canonico_origem', v_canon_origem,
      'manual_financial_adjustment', v_manual, 'manual_adjustment_reason', v_manual_reason,
      'reconciliacao', v_guard),
    v_uid
  );

  RETURN v_receivable_id;
END;

      USING ERRCODE = 'check_violation';

  IF NEW.financial_status IS NULL THEN

  v_qtd_ja_entregue_lote_os numeric;

  -- Se a entrega já está sendo criada/atualizada como cancelada, não validar saldo
  IF NEW.status::text = 'cancelada' THEN
    RETURN NEW;

    AND status::text <> 'cancelada';

    RAISE EXCEPTION 'Saldo físico insuficiente no lote %. Disponível: %, Solicitado: %.',

      USING ERRCODE = 'check_violation';

      USING ERRCODE = 'check_violation';
  END IF;

  RETURN NEW;
END;

  v_uid uuid := auth.uid();
  v_t public.entrega_termos%ROWTYPE;
  v_fp text;
  v_rec text := nullif(trim(coalesce(p_payload->>'recebedor_nome','')), '');
  v_tipo text := coalesce(nullif(p_payload->>'assinatura_tipo',''), 'digital_tela');
  v_ahash text := nullif(p_payload->>'assinatura_hash','');
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'sessao_invalida'; END IF;
  IF NOT public.is_delivery_staff(v_uid) THEN RAISE EXCEPTION 'sem_permissao'; END IF;
  IF coalesce(trim(p_idempotency_key),'') = '' THEN RAISE EXCEPTION 'idempotency_key_obrigatoria'; END IF;
  IF v_rec IS NULL THEN RAISE EXCEPTION 'recebedor_nome_obrigatorio'; END IF;
  IF v_tipo <> 'dispensada' AND v_ahash IS NULL THEN RAISE EXCEPTION 'assinatura_hash_obrigatoria'; END IF;

  SELECT * INTO v_t FROM public.entrega_termos WHERE id = p_termo_id;
  IF NOT FOUND THEN RAISE EXCEPTION 'termo_nao_encontrado'; END IF;
  -- Tenant explícito: ator canônico ativo/pronto (h15) e entrega+cliente da mesma empresa.
  IF NOT platform_private.entrega_termo_tenant_ok(v_t.entrega_id) THEN RAISE EXCEPTION 'termo_nao_encontrado'; END IF;

  PERFORM pg_advisory_xact_lock(hashtextextended('entrega_termo:' || v_t.entrega_id::text, 0));
  SELECT * INTO v_t FROM public.entrega_termos WHERE id = p_termo_id FOR UPDATE;
  -- Revalida após os locks (perfil/papel/empresa podem ter mudado durante a espera).
  IF NOT FOUND OR NOT public.is_delivery_staff(v_uid)
     OR NOT platform_private.entrega_termo_tenant_ok(v_t.entrega_id) THEN
    RAISE EXCEPTION 'termo_nao_encontrado';
  END IF;

  v_fp := encode(extensions.digest(
    v_t.snapshot_hash || '|' || v_rec || '|' || coalesce(v_ahash,'') || '|' || v_tipo, 'sha256'), 'hex');

  -- Replay
  IF v_t.status = 'assinado' THEN
    IF v_t.idempotency_key = p_idempotency_key THEN
      IF v_t.request_fingerprint IS DISTINCT FROM v_fp THEN
        RAISE EXCEPTION 'idempotency_payload_mismatch';
      END IF;
      RETURN jsonb_build_object('success', true, 'replay', true, 'termo_id', v_t.id,
                                'versao', v_t.versao, 'snapshot_hash', v_t.snapshot_hash);
    END IF;
    RAISE EXCEPTION 'termo_ja_assinado';
  END IF;

  IF v_t.status <> 'rascunho' THEN RAISE EXCEPTION 'termo_estado_invalido: %', v_t.status; END IF;

  IF EXISTS (SELECT 1 FROM public.entrega_termos
             WHERE entrega_id = v_t.entrega_id AND idempotency_key = p_idempotency_key AND id <> v_t.id) THEN
    RAISE EXCEPTION 'idempotency_key_reutilizada';
  END IF;

  -- O rascunho deve representar os dados atuais, exceto a hora de geração.
  -- O replay acima permanece imutável e não refaz a assinatura.
  PERFORM 1 FROM public.entregas
   WHERE id = v_t.entrega_id AND status <> 'cancelada'
   FOR SHARE;
  IF NOT FOUND THEN RAISE EXCEPTION 'entrega_invalida'; END IF;

  IF (v_t.snapshot - 'gerado_em') IS DISTINCT FROM
     (public.fn_entrega_build_snapshot_v1(v_t.entrega_id) - 'gerado_em')
     OR v_t.cobertura IS DISTINCT FROM public.fn_entrega_cobertura_v1(v_t.entrega_id) THEN
    RAISE EXCEPTION 'termo_desatualizado: atualize o rascunho antes de assinar';
  END IF;
  IF v_t.snapshot_hash IS DISTINCT FROM
     encode(extensions.digest(v_t.template_versao || '|' || v_t.snapshot::text, 'sha256'), 'hex') THEN
    RAISE EXCEPTION 'termo_integridade_invalida';
  END IF;

  IF v_t.cobertura = 'pendente' AND coalesce((p_payload->>'permitir_pendencia')::boolean, false) IS FALSE THEN
    RAISE EXCEPTION 'cobertura_pendente: conferir todos os itens antes de assinar';
  END IF;

  UPDATE public.entrega_termos SET
    status = 'assinado',
    recebedor_nome = v_rec,
    recebedor_documento = nullif(p_payload->>'recebedor_documento',''),
    recebedor_cargo = nullif(p_payload->>'recebedor_cargo',''),
    recebedor_telefone = nullif(p_payload->>'recebedor_telefone',''),
    assinatura_hash = v_ahash,
    assinatura_tipo = v_tipo,
    assinatura_origem = nullif(p_payload->>'origem',''),
    assinado_em = now(),
    assinado_por = v_uid,
    idempotency_key = p_idempotency_key,
    request_fingerprint = v_fp
  WHERE id = p_termo_id;

  INSERT INTO public.financial_audit_log (entity_type, entity_id, action, new_data, performed_by)
  VALUES ('entrega', v_t.entrega_id, 'DELIVERY_TERM_SIGNED',
          jsonb_build_object('termo_id', p_termo_id, 'versao', v_t.versao,
                             'snapshot_hash', v_t.snapshot_hash, 'cobertura', v_t.cobertura), v_uid);

  RETURN jsonb_build_object('success', true, 'replay', false, 'termo_id', p_termo_id,
                            'versao', v_t.versao, 'snapshot_hash', v_t.snapshot_hash,
                            'cobertura', v_t.cobertura);
END $function$

  v_ratio numeric := 0;
  v_fp text;
  v_item record;
  v_id uuid;
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'sessao_invalida' USING ERRCODE='insufficient_privilege'; END IF;
  IF NOT public.is_delivery_staff(v_uid) THEN
    RAISE EXCEPTION 'sem_permissao' USING ERRCODE='insufficient_privilege';

  IF v_e.status = 'cancelada' THEN RAISE EXCEPTION 'entrega_cancelada'; END IF;

   WHERE entrega_id = p_entrega_id AND status = 'assinado'
   ORDER BY versao DESC LIMIT 1;
  IF NOT FOUND THEN RAISE EXCEPTION 'termo_nao_assinado'; END IF;

  SELECT count(*)::int,
         count(*) FILTER (WHERE status='aceito')::int,
         count(*) FILTER (WHERE status='recusado')::int,
         count(*) FILTER (WHERE status='pendente')::int,
         coalesce(sum(quantidade) FILTER (WHERE status='aceito'),0),
         coalesce(sum(quantidade) FILTER (WHERE status='recusado'),0)
    INTO v_total, v_ok, v_no, v_pend, v_qtd_ok, v_qtd_no
    FROM public.entrega_conferencia_itens WHERE entrega_id = p_entrega_id;

  IF v_total = 0 THEN RAISE EXCEPTION 'conferencia_ausente'; END IF;
  IF v_pend > 0 THEN RAISE EXCEPTION 'conferencia_pendente: % item(ns) sem decisao', v_pend; END IF;

   WHERE entrega_id = p_entrega_id AND status = 'reconciliada';

      FROM public.entrega_conferencia_itens ci
     WHERE ci.entrega_id = p_entrega_id
     ORDER BY ci.id

    IF v_item.status = 'aceito' THEN
      IF v_item.embalagem_id IS NOT NULL THEN
        UPDATE public.estoque_embalagens
           SET status = 'entregue', entregue_em = now(), entregue_por = v_uid,
               entrega_id = p_entrega_id, updated_at = now()
         WHERE id = v_item.embalagem_id

          (embalagem_id, tipo_movimentacao, status_anterior, status_novo, origem, destino,
