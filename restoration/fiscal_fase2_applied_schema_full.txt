-- ================ VIEWS ================
CREATE OR REPLACE VIEW public.v_fiscal_document_reconciliation_pending AS
 SELECT id AS fiscal_invoice_id,
    receivable_id,
    reconciliation_status,
    document_model,
    document_origin,
    status,
    normalized_status
   FROM fiscal_invoices
  WHERE (reconciliation_status = ANY (ARRAY['PENDING'::text, 'DIVERGENT'::text, 'REQUIRES_REVIEW'::text]));
CREATE OR REPLACE VIEW public.v_fiscal_company_readiness AS
 SELECT id,
    configuration_status,
    nfe_enabled,
    nfse_enabled,
    production_enabled,
    fiscal_environment,
    fiscal_provider,
    (((((((cnpj IS NOT NULL))::integer + ((legal_name IS NOT NULL))::integer) + ((tax_regime IS NOT NULL))::integer) + ((crt IS NOT NULL))::integer) + ((primary_cnae IS NOT NULL))::integer) + ((municipality_ibge_code IS NOT NULL))::integer) AS filled_score,
    6 AS max_score,
    (accounting_validated_at IS NOT NULL) AS accounting_validated
   FROM company_fiscal_settings
  WHERE (is_active = true);
CREATE OR REPLACE VIEW public.v_product_fiscal_readiness AS
 SELECT product_id,
    profile_status,
    ((ncm IS NOT NULL) AND (cfop_in_state IS NOT NULL)) AS minimally_classified,
    (accounting_validated_at IS NOT NULL) AS accounting_validated
   FROM product_fiscal_profiles
  WHERE (is_active = true);
CREATE OR REPLACE VIEW public.v_service_fiscal_readiness AS
 SELECT id,
    service_id,
    source_reference,
    profile_status,
    ((lc116_item IS NOT NULL) AND (cnae IS NOT NULL)) AS minimally_classified,
    (accounting_validated_at IS NOT NULL) AS accounting_validated
   FROM service_fiscal_profiles
  WHERE (is_active = true);
CREATE OR REPLACE VIEW public.v_customer_fiscal_readiness AS
 SELECT customer_id,
    profile_status,
    (fiscal_municipality_ibge_code IS NOT NULL) AS has_ibge
   FROM customer_fiscal_profiles;
